\begin{frame}{Why the Higgs-charm coupling?}
  \begin{columns}[T]
    \column{0.58\textwidth}
    \begin{itemize}
      \item Higgs couplings to the 3\textsuperscript{rd} generation (t, b, $\tau$) and to $\mu$ agree with the SM
      \item \textbf{2\textsuperscript{nd}-generation quark Yukawas are the open frontier}: charm is the most accessible
      \begin{itemize}
        \item SM: $\mathcal{B}(\Hcc)\approx 2.9\%$, about 20$\times$ below $\Hbb$
        \item Huge QCD backgrounds, c-jets hard to tell apart from b- and light jets
      \end{itemize}
      \item BSM: flavour-compatible models allow $|\kc|$ of order 3 \\
        {\scriptsize e.g.\ spontaneous flavour violation, vector-like quarks (arXiv:2410.05236)}\\[0.3ex]
        $\Rightarrow$ current sensitivity is \alert{now entering} this region
      \item Is the Higgs mechanism also responsible for the 2\textsuperscript{nd}-generation quark masses?
    \end{itemize}
    \column{0.40\textwidth}
    \centering
    \includegraphics[height=5.6cm]{HIG-21-018_Figure-aux_011.pdf}\\
    {\tiny Rep.\ Prog.\ Phys.\ 89 (2026) 087801 (supplementary); \kc\ from PRL 136 (2026) 011801}\\[0.6ex]
    {\small $\kc = y_\mathrm{c}/y_\mathrm{c}^\mathrm{SM}$ \ {\scriptsize(SM: $\kc=1$)}}
  \end{columns}
\end{frame}

\begin{frame}{H + charm production: \kc\ in production}
  \pubref{JHEP 11 (2025) 060; JHEP 08 (2026) 051}{https://cms-results.web.cern.ch/cms-results/public-results/publications/HIG-24-009/}

      \item \kc\ from ``flat direction'': other H rates kept SM-like

\begin{frame}{Summary}
  \begin{itemize}
    \item The charm Yukawa is the most accessible 2\textsuperscript{nd}-generation quark coupling, and CMS probes it in many complementary ways
    \item Direct searches: \ttH(\Hcc) combined with VH(\Hcc)\\
      $\Rightarrow$ \textbf{$|\kc| < 3.5$ (2.7 exp.)}, the most stringent constraint to date
    \item H+c production probes \kc\ independently of \Hcc: $\gamma\gamma$ + WW give $|\kc|<47$ (51 exp.)
    \item Indirect probes ($\gamma$H, on/off-shell H$\to4\ell$, H$\to$J/$\psi\gamma$) reach $|\kc|\sim3$ under different assumptions; off-shell H$\to4\ell$ removes the assumption on $\kappa_\mathrm{Z,W}$
    \item Sensitivity is now reaching the BSM-motivated region $|\kc|\sim3$
    \item Run 3: new HLT tagging and a measured \ttbar+c$\bar{\mathrm{c}}$ background; HL-LHC projects $|\kc|<1.5$
  \end{itemize}
