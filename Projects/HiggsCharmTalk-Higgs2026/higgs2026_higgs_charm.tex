\begin{frame}{Why the Higgs-charm coupling?}
  \begin{columns}[T]
    \column{0.58\textwidth}
    \begin{itemize}
      \item Higgs couplings to the 3\textsuperscript{rd} generation (t, b, $\tau$) and to $\mu$ are established
      \item \textbf{2\textsuperscript{nd}-generation quark Yukawas are the open frontier}: charm is the most accessible
      \begin{itemize}
        \item SM: $\mathcal{B}(\Hcc)\approx 2.9\%$, about 20$\times$ below $\Hbb$
        \item Huge QCD backgrounds, c-jets hard to tell apart from b- and light jets
      \end{itemize}
      \item BSM: flavour-compatible models allow $|\kc|$ of order 3 \\
        {\scriptsize e.g.\ spontaneous flavour violation, vector-like quarks (arXiv:2410.05236)}\\[0.3ex]
        $\Rightarrow$ current sensitivity is \alert{now entering} this region
      \item Is the Higgs mechanism also responsible for the 2\textsuperscript{nd}-generation quark masses?
    \end{itemize}
    \column{0.40\textwidth}
    \centering
    \begin{tikzpicture}[scale=1.0]
      \draw[dashed,thick] (0,0) -- (1.6,0) node[pos=0,left] {\large H};
      \draw[thick,cmsblue] (1.6,0) -- (2.9,1.1) node[right] {\large c};
      \draw[thick,cmsblue] (1.6,0) -- (2.9,-1.1) node[right] {\large $\bar{\mathrm{c}}$};
      \fill[cmsorange] (1.6,0) circle (3.5pt);
      \node[below=6pt,cmsorange] at (1.6,0) {$y_\mathrm{c}$};
    \end{tikzpicture}\\[1.5em]
    \begin{block}{Coupling modifier}
      \centering
      $\kc = y_\mathrm{c}/y_\mathrm{c}^\mathrm{SM}$\\[0.3ex]
      {\scriptsize ($\kappa$ framework; SM: $\kc=1$)}
    \end{block}
  \end{columns}
\end{frame}

\begin{frame}{The key tool: charm-jet identification}
  \begin{columns}[T]
    \column{0.47\textwidth}
    \begin{itemize}
      \item c-jets sit between b and light jets: shorter lifetime, fewer displaced tracks, softer leptons
      \item Taggers must separate \textbf{c vs.\ light} and \textbf{c vs.\ b} at the same time
      \item Rapid progress through ML on jet constituents:\\
        {\scriptsize DeepJet $\to$ ParticleNet $\to$ (Robust)ParT $\to$ \textbf{UParT}}
      \begin{itemize}
        \item Several-fold gain in c-jet rejection over the Run 2 taggers at similar efficiency
      \end{itemize}
      \item Merged-jet (boosted) \Hcc\ tagging with ParticleNet / GloParT
      \item Calibration in data (W+c, \ttbar\ semileptonic, Z+jets) is a leading systematic
    \end{itemize}
    \column{0.52\textwidth}
    \centering
    \includegraphics[width=\linewidth]{DP-2025-081_sl27_evolution_algo.png}\\
    {\tiny CMS-DP-2025-081: c- and light-jet rejection by tagger generation (13.6\,TeV simulation)}
  \end{columns}
\end{frame}

\begin{frame}{Summary}
  \begin{itemize}
    \item The charm Yukawa is the most accessible 2\textsuperscript{nd}-generation quark coupling, and CMS probes it in many complementary ways
    \item Direct searches: \ttH(\Hcc) added to VH and boosted \Hcc\\
      $\Rightarrow$ \textbf{$|\kc| < 3.5$ (2.7 exp.)}, the most stringent constraint to date
    \item H+c production with $\gamma\gamma$ and WW: $|\kc|<47$ (51 exp.)
    \item Indirect probes ($p_\mathrm{T}^\mathrm{H}$, $\gamma$H, on/off-shell H$\to4\ell$, H$\to$J/$\psi\gamma$) are sensitive to the sign of \kc\ and relax model assumptions
    \item Sensitivity is now reaching the BSM-motivated region $|\kc|\sim3$
    \item Run 3 tools (HLT tagging, UParT, \ttbar+HF measurements) are in place; HL-LHC projects $|\kc|<1.5$
  \end{itemize}
