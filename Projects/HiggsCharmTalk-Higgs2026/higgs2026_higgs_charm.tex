\begin{frame}{Why the Higgs-charm coupling?}
  \begin{columns}[T]
    \column{0.58\textwidth}
    \begin{itemize}
      \item Since the 2012 discovery: couplings to t, b, $\tau$ and $\mu$ measured, all SM-like
      \item \textbf{2\textsuperscript{nd}-generation quark Yukawas are the open frontier}; charm is the most accessible
      \begin{itemize}
        \item $\mathcal{B}(\Hcc)\approx 2.9\%$, about 20$\times$ below $\Hbb$
        \item huge QCD backgrounds; c jets are hard to tell from b and light jets
      \end{itemize}
      \item \textbf{$\kc = y_\mathrm{c}/y_\mathrm{c}^\mathrm{SM}$} \ (SM: $\kc=1$)
      \item Flavour-compatible BSM (e.g.\ spontaneous flavour violation) allows $|\kc|\sim3$: current sensitivity is \alert{entering this region}
      \item Does the Higgs mechanism also give the 2\textsuperscript{nd}-generation quarks their mass?
    \end{itemize}
    \column{0.40\textwidth}
    \centering
    \includegraphics[height=5.9cm]{HIG-21-018_Figure-aux_011.pdf}\\
    {\scriptsize CMS, Rep.\ Prog.\ Phys.\ 89 (2026) 087801 (supplementary)}
  \end{columns}
\end{frame}

\begin{frame}{Direct \Hcc: VH(\Hcc)}
  \pubref{PRL 131 (2023) 061801, CMS-HIG-21-008}{\pubHIG{HIG-21-008}}
  \begin{columns}[T]
    \column{0.32\textwidth}
    \begin{itemize}\small
      \item Leptonic W/Z decays: 0, 1, 2 leptons
      \item Resolved (two small-radius c jets) and merged-jet (one large-radius jet) topologies; ParticleNet c tagging
      \item VZ(Z$\to\ccbar$) observed at 5.7$\sigma$: validates the method
    \end{itemize}
    \begin{block}{VH(\Hcc)}
      {\scriptsize 95\% CL obs.\ (exp.)}\\
      \small $\mu < \obsexp{14}{7.6}$\\
      $1.1 < |\kc| < 5.5$ \ {\color{cmsgrey}($<3.4$)}\\
      {\scriptsize obs.\ excludes $\kc=1$: upward fluctuation, $\hat\mu = 7.7$ \tobackup}
    \end{block}
    \column{0.33\textwidth}
    \centering
    \includegraphics[width=\linewidth]{HIG-21-008_Figure_003.pdf}\\

\begin{frame}{H + charm production: \kc\ in production}
  \pubref{JHEP 11 (2025) 060, CMS-HIG-23-010}{\pubHIG{HIG-23-010}}
  \begin{itemize}\small
    \item \kc\ enters the \textbf{production} vertex, independent of the \Hcc\ decay: clean H decay + a c-tagged jet
    \item $\sigma_{\mathrm{cH}} = 90$\,fb: $\sigma\mathcal{B}\approx 0.2$\,fb ($\gamma\gamma$), $\approx 20$\,fb (WW)
    \item Challenges: \textbf{\kc-independent H+c} from g$\to\ccbar$ splitting in ggH (same final state); soft, often forward c jets; flavour-scheme modelling ($\to$ T.~Bevilacqua, next talk)
    \item \textbf{``Flat direction''}: \kc\ is varied while all other Higgs rates stay at their SM values, so only cH production changes
  \end{itemize}

\begin{frame}{Indirect constraints on \kc}
  {\hfill\scriptsize\href{\pubHIG{HIG-23-011}}{\color{cmsblue}PRD 112 (2025) 112001}; \href{\pubHIG{HIG-25-003}}{\color{cmsblue}arXiv:2607.23352}; \href{\pubHIG{SMP-22-012}}{\color{cmsblue}PLB 865 (2025) 139462}}
  \begin{columns}[T]
    \column{0.55\textwidth}
    \begin{itemize}\small
      \item Large \kc\ adds $\ccbar\to$H production and increases $\Gamma_\mathrm{H}$: on-shell H$\to4\ell$ rates constrain it
      \item \textbf{On-shell H$\to4\ell$} ($\gamma$H paper), $|\kappa_\mathrm{Z}=\kappa_\mathrm{W}|\le1$:\\ \kc\ $\in [-3.7, 3.2]$ {\color{cmsgrey}($[-3.8,3.3]$)}
      \item \new\ \textbf{Off-shell + on-shell H$\to4\ell$}: the off-shell rate breaks the $\kappa_V$--$\Gamma_\mathrm{H}$ degeneracy
      \begin{itemize}
        \item $|\kappa_V|\le1$: \kc\ $\in [-3.3, 2.8]$ {\color{cmsgrey}($[-3.5,2.9]$)}
        \item $\kappa_\mathrm{Z}$, $\kappa_\mathrm{W}$ free: \kc\ $\in [-8.6, 7.5]$ {\color{cmsgrey}($[-9.6,8.3]$)}
        \item {\scriptsize width measurement: F.~Errico's talk}
      \end{itemize}
      \item H$\to$J/$\psi\gamma$: $-166 < \kc < 208$ {\scriptsize(J.~Bierkens' talk)}
    \end{itemize}
    \column{0.42\textwidth}
    \centering
    \includegraphics[width=0.92\linewidth]{HIG-25-003_Figure_008-d.pdf}\\
    {\scriptsize \kc\ scans, solid = observed, dashed = expected}
  \end{columns}
  \takeaway{With $|\kappa_V|\le1$, H$\to4\ell$ matches the direct searches, under different assumptions \ \tobackup}

\begin{frame}{Summary}
  \begin{itemize}
    \item The charm Yukawa is the most accessible 2\textsuperscript{nd}-generation quark coupling; CMS probes it in decay, in production and indirectly
    \item \ttH(\Hcc) + VH(\Hcc): \textbf{$|\kc| < 3.5$ (2.7 exp.)}, the most stringent direct constraint and the best expected sensitivity
    \item H+c production: $|\kc|<47$ (51 exp.), independent of the \Hcc\ decay
    \item Off-shell + on-shell H$\to4\ell$: $\kc\in[-3.3,2.8]$ with $|\kappa_V|\le1$ (about $\pm8$ without)
    \item All \kc\ results are Run 2; Run 3 brings higher trigger efficiency and a measured \ttbar+\ccbar\ background
    \item \textbf{Does the Higgs give charm its mass?} Today $|\kc|\lesssim3$--3.5, entering BSM-motivated territory; the HL-LHC is expected to reach $|\kc|<1.5$
  \end{itemize}

\end{frame}

\begin{frame}{Likely questions}
  \small
  \begin{itemize}
    \item \textbf{How large is the VH excess?} Best fit $\mu_\mathrm{VH} = 7.7^{+3.8}_{-3.5}$, about 2 standard deviations above the SM. This is why VH alone excludes $|\kc|<1.1$ and why \ttH\ alone (3.0) beats the combination (3.5) on observed
    \item \textbf{What limits the direct searches?} Statistics first. \ttH(\ccbar): statistical 74\% of the total uncertainty, then \ttbar+c and \ttbar+light modelling ($\approx$32\%, $\approx$29\%). VH(\ccbar): simulation sample size ($\approx$37\%), c-tagging efficiency ($\approx$23\%), theory ($\approx$22\%)
    \item \textbf{And ATLAS?} VBF + VH(\Hcc) combination (arXiv:2511.21911): $|\kc| < 4.7$ (3.9 exp.)
    \item \textbf{Why is cH an order of magnitude weaker than \Hcc?} cH $\sigma\mathcal{B}$ is tiny ($\approx$0.2\,fb in $\gamma\gamma$) and ggH produces the same H+c final state independently of \kc; in \Hcc, \kc\ changes the branching fraction of Higgs bosons from much larger production modes (VH, \ttH)
    \item \textbf{What is fixed in each \kc\ number?} See the assumptions column of the summary table (slide 12)
  \end{itemize}
\end{frame}
