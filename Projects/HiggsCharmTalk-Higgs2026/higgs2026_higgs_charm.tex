\begin{frame}{Why the Higgs-charm coupling?}
  \begin{columns}[T]
    \column{0.58\textwidth}
    \begin{itemize}
      \item Since the 2012 discovery: couplings to t, b, $\tau$ measured, evidence for $\mu$ {\scriptsize(JHEP 01 (2021) 148)} \tobackup; all SM-like
      \item \textbf{2\textsuperscript{nd}-generation quark Yukawas are the open frontier}; charm is the most accessible
      \begin{itemize}
        \item $\mathcal{B}(\Hcc)\approx 2.9\%$, about 20$\times$ below $\Hbb$
        \item huge QCD backgrounds; c jets are hard to tell from b and light jets
      \end{itemize}
      \item \textbf{$\kc = y_\mathrm{c}/y_\mathrm{c}^\mathrm{SM}$} \ (SM: $\kc=1$)
      \item Flavour-compatible BSM (e.g.\ spontaneous flavour violation) allows $|\kc|\sim3$: current sensitivity is \alert{entering this region}
      \item Does the Higgs mechanism also give the 2\textsuperscript{nd}-generation quarks their mass?
    \end{itemize}
    \column{0.40\textwidth}
    \centering
    \includegraphics[height=5.9cm]{HIG-21-018_Figure-aux_011.pdf}\\
    {\scriptsize CMS, Rep.\ Prog.\ Phys.\ 89 (2026) 087801 (suppl.); \kc\ point: separate \ttH+VH result, not in the fit}
  \end{columns}
\end{frame}

\begin{frame}{Direct \Hcc: VH(\Hcc)}
  \pubref{PRL 131 (2023) 061801, CMS-HIG-21-008}{\pubHIG{HIG-21-008}}
  \begin{columns}[T]
    \column{0.37\textwidth}
    \begin{itemize}\footnotesize
      \item W/Z $\to$ 0, 1, 2 leptons; main backgrounds V+jets and \ttbar, normalised in control regions
      \item \textbf{Resolved} ($p_\mathrm{T}^\mathrm{H}<300$\,GeV, $>$95\% of signal): two DeepJet c-tagged jets; fit to a BDT
      \item \textbf{Merged} (large-R jet $p_\mathrm{T}>300$\,GeV): $<$5\% of signal but less background; ParticleNet \ccbar\ tagging (3 purity categories) and mass regression; fit to $m(\mathrm{H_{cand}})$
      \item Z$\to\ccbar$ in VZ: \textbf{first observation at a hadron collider}, 5.7$\sigma$
    \end{itemize}
    \column{0.30\textwidth}
    \centering
    \includegraphics[width=\linewidth]{HIG-21-008_Figure_003.pdf}\\
    {\scriptsize Merged-jet only; bottom: VZ (grey, $\approx$90\,GeV) and VH (red) after background subtraction}
    \column{0.30\textwidth}
    \centering
    \includegraphics[width=0.85\linewidth]{HIG-21-008_Figure_004.pdf}\\
    {\scriptsize $\mu$ limits per channel (inner band 68\%)}\\[0.3ex]
    \begin{block}{VH(\Hcc)}

\begin{frame}{H + charm production: \kc\ in production}
  \pubref{JHEP 11 (2025) 060, CMS-HIG-23-010}{\pubHIG{HIG-23-010}}
  \begin{itemize}\small
    \item \kc\ enters the \textbf{production} vertex, independent of the \Hcc\ decay: clean H decay + a c-tagged jet
    \item $\sigma_{\mathrm{cH}} = 90$\,fb: $\sigma\mathcal{B}\approx 0.2$\,fb ($\gamma\gamma$), $\approx 20$\,fb (WW)
    \item Challenges: \textbf{\kc-independent H+c} from g$\to\ccbar$ splitting in ggH (same final state); soft, often forward c jets; flavour-scheme modelling ($\to$ T.~Bevilacqua, next talk)
    \item \textbf{``Flat direction''}: \kc\ is varied while all other Higgs rates stay at their SM values, so only cH production changes
  \end{itemize}

\begin{frame}{cH(H$\to\gamma\gamma$)}
  \pubref{JHEP 11 (2025) 060, CMS-HIG-23-010}{\pubHIG{HIG-23-010}}
  \begin{columns}[T]
    \column{0.50\textwidth}
    \begin{itemize}\small
      \item First search for cH with H$\to\gamma\gamma$, 138\fbinv
      \item Two BDTs, trained without c-tag inputs: cH vs.\ ggH (largest resonant background), cH vs.\ continuum $\gamma\gamma$+jets
      \item 9 categories per year (27 in total); signal from fits to $m_{\gamma\gamma}$
      \item Plot: the 125\,GeV peak is fully explained by the other Higgs processes (mostly ggH); nothing is left for cH, best fit $\hat\mu_\mathrm{cH} = -230^{+170}_{-251}$
    \end{itemize}
    \begin{block}{cH(H$\to\gamma\gamma$)}
      {\scriptsize 95\% CL obs.\ (exp.)}\\
      \small $\mu_{\mathrm{cH}} < \obsexp{243}{355}$, \ $|\kc| < \obsexp{38}{73}$
    \end{block}
    \column{0.46\textwidth}
    \centering
    \includegraphics[height=5.3cm]{HIG-23-010_Figure_006.pdf}\\
    {\scriptsize $m_{\gamma\gamma}$, all categories, weighted by S/(S+B)}

\begin{frame}{Indirect constraints on \kc}
  {\hfill\scriptsize\href{\pubHIG{HIG-23-011}}{\color{cmsblue}PRD 112 (2025) 112001}; \href{\pubHIG{HIG-25-003}}{\color{cmsblue}arXiv:2607.23352}; \href{\pubHIG{SMP-22-012}}{\color{cmsblue}PLB 865 (2025) 139462}}
  \begin{columns}[T]
    \column{0.55\textwidth}
    \begin{itemize}\small
      \item Large \kc\ adds $\ccbar\to$H production and increases $\Gamma_\mathrm{H}$: on-shell H$\to4\ell$ rates constrain it\\
        {\scriptsize $\kappa_V$ ($\kappa_\mathrm{Z}$, $\kappa_\mathrm{W}$): HZZ, HWW coupling modifiers. Also (\kb, \kc) from the Higgs $p_\mathrm{T}$ spectrum \tobackup}
      \item \textbf{On-shell H$\to4\ell$}, $|\kappa_\mathrm{Z}=\kappa_\mathrm{W}|\le1$:\\ \kc\ $\in [-3.7, 3.2]$ {\color{cmsgrey}($[-3.8,3.3]$)}
      \item \new\ \textbf{Off-shell + on-shell H$\to4\ell$}: the off-shell rate breaks the $\kappa_V$--$\Gamma_\mathrm{H}$ degeneracy
      \begin{itemize}
        \item $|\kappa_V|\le1$: \kc\ $\in [-3.3, 2.8]$ {\color{cmsgrey}($[-3.5,2.9]$)}
        \item $\kappa_\mathrm{Z}$, $\kappa_\mathrm{W}$ free: \kc\ $\in [-8.6, 7.5]$ {\color{cmsgrey}($[-9.6,8.3]$)}
        \item {\scriptsize width measurement: F.~Errico's talk (Tue)}
      \end{itemize}
      \item H$\to$J/$\psi\gamma$: $-166 < \kc < 208$ {\scriptsize(J.~Bierkens' talk, Thu)}
    \end{itemize}
    \column{0.42\textwidth}
    \centering
    \includegraphics[width=0.82\linewidth]{HIG-25-003_Figure_008-d.pdf}\\
    {\scriptsize arXiv:2607.23352 Fig.\ 8: blue on-shell and red on+off-shell with $|\kappa_\mathrm{Z}=\kappa_\mathrm{W}|\le1$; green on-shell and black on+off-shell with $\kappa_\mathrm{Z}\neq\kappa_\mathrm{W}$; solid obs., dashed exp. Text intervals: Table 3 (no custodial symmetry)}
  \end{columns}
  \takeaway{With $|\kappa_V|\le1$, H$\to4\ell$ matches the direct searches, under different assumptions \ \tobackup}
\end{frame}

\begin{frame}{Summary}
  \begin{itemize}
    \item The charm Yukawa is the most accessible 2\textsuperscript{nd}-generation quark coupling; CMS probes it in decay, in production and indirectly
    \item \ttH(\Hcc) + VH(\Hcc): \textbf{$|\kc| < \obsexp{3.5}{2.7}$}, the most stringent direct constraint and the best expected sensitivity
    \item H+c production: $|\kc|<\obsexp{47}{51}$, independent of the \Hcc\ decay
    \item Off-shell + on-shell H$\to4\ell$: $\kc\in[-3.3,2.8]$ with $|\kappa_V|\le1$ (about $\pm8$ without)
    \item All \kc\ results are Run 2; Run 3 searches start with higher trigger efficiency and a \ttbar+\ccbar\ background already measured in Run 2
    \item \textbf{Does the Higgs give charm its mass?} Today $|\kc|\lesssim3$--3.5, entering BSM-motivated territory; HL-LHC expected $|\kc|<1.5$: approaching, not yet reaching, sensitivity to $\kc=0$
  \end{itemize}

\begin{frame}{Likely questions}
  \footnotesize
  \begin{itemize}
    \item \textbf{How large is the VH excess?} Best fit $\mu_\mathrm{VH} = 7.7^{+3.8}_{-3.5}$, about 1.9$\sigma$ above $\mu=1$ (estimated from the quoted uncertainty; the paper calls it a small upward fluctuation). This is why VH alone excludes $|\kc|<1.1$ and why \ttH\ alone (3.0) beats the combination (3.5) on observed
    \item \textbf{What limits the direct searches?} Statistics first. \ttH(\ccbar): statistical 74\% of the total uncertainty, then \ttbar+c and \ttbar+light modelling ($\approx$32\%, $\approx$29\%). VH(\ccbar): simulation sample size ($\approx$37\%), c-tagging efficiency ($\approx$23\%), theory ($\approx$22\%)
    \item \textbf{And ATLAS?} VBF + VH(\Hcc) (arXiv:2511.21911): $|\kc| < 4.7$ (3.9 exp.), same assumptions as CMS (other couplings SM, no BSM decays); its VBF part includes 51.5\fbinv\ of Run 3 data
    \item \textbf{Why is cH an order of magnitude weaker than \Hcc?} cH $\sigma\mathcal{B}$ is tiny ($\approx$0.2\,fb in $\gamma\gamma$) and ggH produces the same H+c final state independently of \kc; in \Hcc, \kc\ changes the branching fraction of Higgs bosons from much larger production modes (VH, \ttH)
    \item \textbf{Why is cH($\gamma\gamma$) obs.\ (38) much better than exp.\ (73)?} The best fit is negative, $\hat\mu_\mathrm{cH} = -230^{+170}_{-251}$
    \item \textbf{Why no \kc\ from boosted \Hcc?} The paper quotes only $\mu$; the result is ggH-dominated
    \item \textbf{How much of the Run 3 trigger gain is the tagger?} Part of it is rate budget: 10\,Hz in 2018 vs.\ 60, 115, 170\,Hz in 2022, 2023, 2024
    \item \textbf{Are the H$\to4\ell$ intervals sign-sensitive?} Not really: the small asymmetry comes from charm-loop interference in ggH and H$\to$gg; the constraint is essentially on $\kc^2$
  \end{itemize}
\end{frame}
