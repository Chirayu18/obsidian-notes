\section{Jet clustering, and flashjet}

\begin{frame}{Jet clustering: the workhorse, and its cost}
\begin{columns}[T]
\begin{column}{0.53\textwidth}
Sequential recombination --- \akt, $k_t$, Cambridge--Aachen, as implemented in
\textbf{FastJet} --- is the LHC's standard jet definition. It is
\textbf{infrared and collinear safe}, which is why it survived, and it is
\textbf{$O(N\log N)$ but strictly sequential}: each merge depends on the one
before it.
\medskip

That sequential structure is exactly what has kept it on the \textbf{CPU} while
the rest of the pipeline moved to accelerators.
\medskip

\smallnote{And it is rarely run once. The same event is clustered again for every
jet definition, every radius, every systematic variation --- and, in a training
loop, once per epoch.}
\end{column}
\begin{column}{0.47\textwidth}
\begin{block}{How often clustering is re-run}
\small
\begin{tabular}{@{}lr@{}}
\toprule
context & per event\\
\midrule
reconstruction & a few\\
\smallnote{small-$R$, large-$R$, trimmed} & \\
JES/JER systematics & $\mathcal{O}(10\text{--}100)$\\
substructure (2nd pass) & $\times 2$\\
$R$- / grooming scan & $\mathcal{O}(10)$\\
\textbf{ML training} & \textbf{1 / epoch}\\
\bottomrule
\end{tabular}
\end{block}
\smallnote{Each row multiplies the one above it. The cost is not one clustering
--- it is the same event clustered many times over.}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{flashjet}
\begin{columns}[T]
\begin{column}{0.54\textwidth}
\textbf{What it is.} A GPU-native implementation of the generalised-$k_t$ family
--- \akt{} ($p\!=\!-1$), $k_t$ ($p\!=\!+1$), Cambridge--Aachen ($p\!=\!0$) ---
written in \textbf{Triton/PyTorch}.
\medskip

\textbf{How it works.} A batch of jets or events is clustered
\textbf{in device memory}, and the data never returns to the host during
clustering. The merge history is recorded \textbf{in-kernel}, so the full
clustering tree comes back with the jets.
\medskip

\textbf{Backends}, dispatched automatically by problem size:
\begin{itemize}
  \item a fused register kernel for small $N$
  \item a scratch nearest-neighbour Triton kernel for large $N$
  \item a pure-torch fallback --- so the same call runs on CPU
\end{itemize}
\end{column}
\begin{column}{0.46\textwidth}
\begin{block}{Design choices}
\begin{itemize}
  \item \textbf{batched}: many jets at once, ragged input via a mask
  \item \textbf{history kept}: substructure becomes an array read, not a second
        clustering pass
  \item \textbf{one entry point}: identical call on CPU and GPU
\end{itemize}
\end{block}
\smallskip
\begin{block}{Status}
\small A \textcolor{fjorange}{\textbf{prototype}} --- validated, but a
Python/Triton library, not integrated into experiment software.
\end{block}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{How sequential recombination works}
\begin{center}
\includegraphics[width=0.94\linewidth]{fig/clustering_steps.png}
\end{center}
\vspace{-0.4em}
\begin{columns}[T]
\begin{column}{0.56\textwidth}
\smallnote{Repeat until nothing is left: if the smallest distance is a
$d_{ij}$, merge that pair; if it is a $d_{iB}$, call that a jet and remove it.}
\end{column}
\begin{column}{0.44\textwidth}
\smallnote{One exponent $p$ selects the algorithm --- the \emph{same} kernel
serves all three.}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{One exponent, three algorithms}
\begin{center}
\begin{tabular}{@{}llll@{}}
\toprule
$p$ & algorithm & merges first & gives\\
\midrule
$-1$ & \akt & around the \textbf{hardest} particles & rigid, circular jets\\
$0$ & Cambridge--Aachen & the \textbf{closest} pair & angular-ordered tree\\
$+1$ & $k_t$ & the \textbf{softest} pair & $k_t$-ordered tree\\
\bottomrule
\end{tabular}
\end{center}
\vspace{0.3em}
\begin{columns}[T]
\begin{column}{0.46\textwidth}
\includegraphics[width=0.98\linewidth]{fig/jet_areas.png}
\end{column}
\begin{column}{0.54\textwidth}
\textbf{The ordering is not a detail --- it is what the substructure reads.}
\smallskip
\begin{itemize}
  \item \akt{} gives the \emph{jets}: catchment areas $\to \pi R^2$ --- the
        defining rigid-cone result, from flashjet's own clustering.
  \item \textbf{C/A} orders by angle $\Rightarrow$ the natural tree for
        \textbf{grooming} and the \textbf{Lund plane}.
  \item \textbf{$k_t$} orders by relative transverse momentum $\Rightarrow$ the
        natural tree for \textbf{exclusive subjets}.
\end{itemize}
\smallnote{So flashjet keeps whichever history the observable needs --- and all
three cost the same.}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{It returns the tree, not just the jets}
Clustering a batch returns \textbf{two} things --- the jets, and the record of how
they were built.
\medskip

\begin{center}
\small
\begin{tabular}{@{}lll@{}}
\toprule
\textbf{field} & \textbf{shape} & \textbf{what it holds}\\
\midrule
\texttt{jet\_idx} & $(B,N)$ & which jet each particle ended up in\\
\texttt{n\_jets} & $(B,)$ & how many jets were found\\
\addlinespace
\texttt{hist\_p1}, \texttt{hist\_p2} & $(B,N)$ & \textbf{which pair merged}, at each step\\
\texttt{hist\_child} & $(B,N)$ & what they merged \emph{into}\\
\texttt{hist\_d} & $(B,N)$ & the distance $d_{ij}$ at that step\\
\bottomrule
\end{tabular}
\end{center}
\medskip

\begin{block}{Why the second half matters}
\small The bottom three rows are the \textbf{complete binary merge tree}. Groomed
mass, $z_g$, $R_g$, Lund coordinates and exclusive subjets are all
\textbf{coordinates on that tree} --- so they become array reads, with
\textbf{no second clustering pass}.
\end{block}
\end{frame}

\begin{frame}[fragile]{Using it}
\begin{Verbatim}[fontsize=\footnotesize,frame=single,rulecolor=\color{fjgray}]
import flashjet

out = flashjet.cluster(p4, mask, R=1.0, algorithm="antikt")
\end{Verbatim}
\vspace{-0.3em}
{\footnotesize
\begin{tabular}{@{}lll@{}}
\toprule
\textbf{argument} & \textbf{type} & \\
\midrule
\texttt{p4} & $(B,N,4)$ tensor & $(p_x,p_y,p_z,E)$; CPU or CUDA\\
\texttt{mask} & $(B,N)$ bool & marks real constituents in a padded batch\\
\texttt{R} & float & jet radius\\
\texttt{algorithm} & str & \texttt{"antikt"} $\vert$ \texttt{"kt"} $\vert$ \texttt{"cambridge"}\\
\texttt{p} & float & generalised-$k_t$ exponent (overrides \texttt{algorithm})\\
\bottomrule
\end{tabular}}
\medskip

\begin{Verbatim}[fontsize=\footnotesize,frame=single,rulecolor=\color{fjgray}]
out.n_jets                                    # (B,)     jets found
out.jets_p4(p4)                               # (B,J,4)  jet four-momenta
out.splitting_scales()                        #          sqrt(d12), sqrt(d23), ...
out.exclusive_jets(n_jets=3)                  # F1       exclusive kt subjets
out.groomed_jets(p4, R, z_cut=0.1, beta=0.0)  # F2       soft drop / mMDT
out.lund_coordinates(p4, R)                   # F3       primary Lund plane
\end{Verbatim}
\smallnote{The same call runs unchanged on CPU and GPU --- the backend is chosen
from the problem size, not by the caller.}
\end{frame}

\begin{frame}{What is implemented}
\begin{columns}[T]
\begin{column}{0.5\textwidth}
\textbf{Clustering}
\begin{itemize}
  \item generalised-$k_t$: \akt, $k_t$, C/A
  \item batched, ragged input via a mask
  \item three backends, auto-dispatched by size
  \item merge history recorded in-kernel
  \item jet regime \emph{and} event regime
\end{itemize}
\end{column}
\begin{column}{0.5\textwidth}
\textbf{Substructure} --- reads of the history
\begin{itemize}
  \item \textbf{F1} exclusive $k_t$ subjets
  \item \textbf{F2} soft drop / mMDT / mass-drop
  \item \textbf{F3} primary Lund coordinates
\end{itemize}
\smallskip
\smallnote{All three are pure-torch post-reads: no kernel changes, CPU/GPU
identical, negligible cost.}
\end{column}
\end{columns}
\vspace{0.8em}
\begin{block}{It is differentiable}
\small \texttt{jets\_p4} is a \texttt{scatter\_add} of the constituents onto their
jets, so \textbf{gradients flow through the clustering} --- the jet four-momenta
can sit inside a loss without a custom backward pass.
\end{block}
\end{frame}

% ------------------------------------------------------------
\begin{frame}{F1 --- exclusive subjets from the $k_t$ history}
\begin{columns}[T]
\begin{column}{0.42\textwidth}
\textbf{What it does.} Undo the last merges of the sequence to expose exactly
$n$ subjets --- or stop at a $d_{cut}$.
\smallskip

\textbf{How.} The history is already a binary tree. Walk back up it $n-1$ steps;
no reclustering.
\smallskip

\textbf{Why $k_t$.} $k_t$ merges the softest pair first, so the \emph{last}
merges are the hard prongs --- exactly what a 2-prong decay leaves behind.
\smallskip

\smallnote{The splitting scales $\sqrt{d_{12}}, \sqrt{d_{23}},\dots$ fall out of
the same array.}
\end{column}
\begin{column}{0.58\textwidth}
\includegraphics[width=\linewidth]{fig/kt_observables.png}\\
\smallnote{Toy W-like (2-prong) vs QCD-like fat jets. \textbf{Left:} the $k_t$
splitting scale $\sqrt{d_{12}}$ separates cleanly. \textbf{Right:} the momentum
balance $z$ of the two exclusive subjets --- the W sits near $z\!\sim\!0.4$,
QCD spreads to small $z$.}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{F2 --- soft drop / mMDT grooming from the C/A history}
\begin{center}
\includegraphics[width=0.95\linewidth]{fig/softdrop_walk.png}
\end{center}
\vspace{-0.3em}
\begin{columns}[T]
\begin{column}{0.5\textwidth}
\smallnote{Walk down the C/A tree along the harder branch, discarding the softer
prong until the momentum-sharing condition is met. $\beta=0$ recovers mMDT.}
\end{column}

\end{column}

\end{frame}

\end{column}
\begin{column}{0.5\textwidth}
\textbf{Method}
\begin{enumerate}
  \item extract constituents once, save;
  \item cluster on GPU, time it, save the \emph{same} events;
  \item cluster those saved events with FastJet;
  \item \textbf{check they agree} --- else the timing is meaningless.
\end{enumerate}
\smallskip
\begin{block}{}
\small Both clusterers read the \textbf{same saved file}. Speedups are quoted
against the \textbf{vectorised} FastJet interface --- the faster, fairer baseline.
\end{block}
\end{column}
\end{columns}

\section{Conclusions}

\begin{frame}{Summary}
\begin{enumerate}
  \item \textbf{Clusters on the GPU and keeps the full merge history}, so
  substructure --- exclusive subjets, soft drop, Lund coordinates --- comes free
  as array reads.
  \medskip
  \item \textbf{Gives the same jets as FastJet.} 150/150 grid points at 100\,\%
  $n_{\rm jets}$ agreement, across \akt/$k_t$/C-A, $R=0.4$--$1.2$ and 24--123
  constituents per jet. $k_t$ and C/A validated on real data for the first time.
  \medskip
  \item \textbf{Reproduces the experiment's own reconstructed jets} exactly, at
  $\sim$600 clusters per event.
  \medskip
  \item \textbf{39--99$\times$ faster} than vectorised FastJet, and the advantage
  \emph{grows} with jet complexity. Cost is independent of $R$ and of algorithm.
  \medskip
  \item Still a \textcolor{fjorange}{\textbf{prototype}}, with meaningful headroom
  left.
\end{enumerate}
\end{frame}
