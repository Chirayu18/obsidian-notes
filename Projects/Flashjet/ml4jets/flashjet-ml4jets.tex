\begin{frame}{Summary --- what this talk shows}
\textbf{A GPU implementation of the generalised-$k_t$ algorithms} (\akt, $k_t$,
Cambridge--Aachen) that clusters batches directly in device memory and keeps the
\textbf{complete merge history} --- so substructure follows as cheap array reads,
not a second clustering pass.
\medskip

\textbf{Validation ladder}, each rung closed against an independent reference:

\begin{center}
\small
\begin{tabular}{@{}lll@{}}
\toprule
\textbf{rung} & \textbf{reference} & \textbf{headline}\\
\midrule
unit tests & independent NumPy tree-walks & 129 passed, incl.\ GPU paths\\
analytic closures & leading-log predictions, toy shower & $z_g$ on the $1/z$ curve;
areas; $\beta$-ordering\\
\textbf{vs FastJet} & same events, same job & \textbf{150/150} points,
\textbf{100\,\%} $n_{\rm jets}$ agreement\\
\textbf{vs the experiment} & ATLAS's own reconstructed jets & \textbf{100\,\%}
match, $\sim$600 clusters/event\\
speed & vectorised FastJet, identical input & \textbf{39--99$\times$}, growing
with multiplicity\\
\bottomrule
\end{tabular}
\end{center}
\medskip

\begin{block}{}
\small \textbf{flashjet reproduces FastJet --- and the experiment's own jets ---
while running roughly two orders of magnitude faster than per-jet CPU
clustering.} It is a \textcolor{fjorange}{\textbf{prototype}}: validated, but not
yet inside any experiment's software.
\end{block}
\end{frame}

% ============================================================
\section{What flashjet is}

\begin{frame}{Clustering is the CPU step in a GPU pipeline}
\begin{columns}[T]
\begin{column}{0.56\textwidth}
Jet clustering sits on the critical path of nearly every jet-level ML pipeline
--- and is almost always done \textbf{off the accelerator}.
\medskip

Four-momenta are copied to the host, clustered, copied back.
\medskip

That round trip dominates when
\begin{itemize}
  \item training revisits the same events for \textbf{many epochs},
  \item jets are \textbf{reclustered} under many variations,
  \item clustering must sit \textbf{inside} a differentiable pipeline.
\end{itemize}
\end{column}
\begin{column}{0.44\textwidth}
\begin{block}{flashjet}
Implements \akt, $k_t$ and C/A \textbf{natively on the GPU}.
\smallskip

Data never leaves device memory during clustering.
\end{block}
\smallnote{Triton/PyTorch. One entry point; runs unchanged on CPU and GPU.}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{It returns the tree, not just the jets}
\begin{columns}[T]
\begin{column}{0.5\textwidth}
Clusters a \textbf{batch of jets or events at once}, and returns \textbf{two}
things:
\begin{enumerate}
  \item the particle $\rightarrow$ jet assignment;
  \item the \textbf{complete merge history} --- which pair merged, into what, at
        what distance.
\end{enumerate}
\end{column}
\begin{column}{0.5\textwidth}
\begin{block}{Why the history matters}
Groomed mass, $z_g$, $R_g$, Lund coordinates, exclusive subjets become
\textbf{reads of that tree}.
\smallskip

No second clustering pass.
\end{block}
\end{column}
\end{columns}
\vspace{1em}
\begin{center}
\small
\begin{tabular}{@{}ll@{}}
\toprule
\texttt{jet\_idx}, \texttt{n\_jets} & particle-to-jet assignment\\
\texttt{hist\_p1}, \texttt{hist\_p2}, \texttt{hist\_child} & which pair merged into what\\
\texttt{hist\_d} & the $d_{ij}$ at each merge step\\
\bottomrule
\end{tabular}
\end{center}
\end{frame}

\begin{frame}[fragile]{One entry point}
\begin{Verbatim}[fontsize=\small,frame=single,rulecolor=\color{fjgray}]
out = flashjet.cluster(p4, mask, R=1.0, algorithm="antikt")
#   p4        (B, N, 4) torch tensor   |  mask  (B, N) bool
#   algorithm "antikt" | "kt" | "cambridge"

out.n_jets                                      # jets found
out.jets_p4(p4)                                 # jet four-momenta
out.exclusive_jets(n_jets=3)                    # F1
out.groomed_jets(p4, R, z_cut=0.1, beta=0.0)    # F2  soft drop / mMDT
out.lund_coordinates(p4, R)                     # F3  primary Lund plane
\end{Verbatim}
\medskip
\begin{block}{}
\small Same call on CPU and GPU. \texttt{jets\_p4} is a \texttt{scatter\_add} of
constituents, so \textbf{gradients flow through the clustering}.
\end{block}
\end{frame}

\begin{frame}{What is implemented}
\begin{columns}[T]
\begin{column}{0.5\textwidth}
\textbf{Clustering}
\begin{itemize}
  \item generalised-$k_t$: \akt, $k_t$, C/A
  \item batched, ragged input via a mask
  \item three backends, auto-dispatched by size
  \item merge history recorded in-kernel
  \item jet regime \emph{and} event regime
\end{itemize}
\end{column}
\begin{column}{0.5\textwidth}
\textbf{Substructure} --- reads of the history
\begin{itemize}
  \item \textbf{F1} exclusive $k_t$ subjets
  \item \textbf{F2} soft drop / mMDT / mass-drop
  \item \textbf{F3} primary Lund coordinates
\end{itemize}
\smallskip
\smallnote{All three are pure-torch post-reads: no kernel changes, CPU/GPU
identical, negligible cost.}
\end{column}
\end{columns}
\vspace{0.8em}
\begin{block}{Status}
\small A \textcolor{fjorange}{\textbf{prototype}} --- the physics is validated and
the speedups are real, but it is a Python/Triton library, \textbf{not} integrated
into any experiment's reconstruction framework.
\end{block}
\end{frame}

\begin{frame}{The benchmark}
\begin{columns}[T]

\textbf{Data} --- ATLAS Open Data, freely presentable

\smallskip
\begin{tabular}{@{}lrr@{}}

\midrule

\bottomrule

\smallskip

\smallnote{Large-$R$ jets with calorimeter-cluster constituents.}
\end{column}
\begin{column}{0.5\textwidth}
\textbf{Method}
\begin{enumerate}
  \item extract constituents once, save;
  \item cluster on GPU, time it, save the \emph{same} events;
  \item cluster those saved events with FastJet;
  \item \textbf{check they agree} --- else the timing is meaningless.
\end{enumerate}
\smallskip
\smallnote{Both clusterers read the same saved file. Speedups are quoted against
the \textbf{vectorised} FastJet interface --- the faster, fairer baseline.}
\end{column}
\end{columns}

\end{block}

roughly \emph{two orders of magnitude}.}
\end{frame}

\begin{frame}{Caveats}
\begin{itemize}
  \item \textbf{The CPU baseline is FastJet's Python binding.} The production tool
  is C++. Until that is measured, the honest caveat is \emph{``some of this could
  be Python overhead''} --- \textcolor{fjred}{the single most valuable missing
  test}.
  \medskip
  \item \textbf{Large-$R$ jets.} Boosted fat jets, $\sim$50--120 constituents ---
  not small-$R$ jets.
  \medskip
  \item \textbf{Batch regime.} The advantage comes from clustering \emph{many}
  jets at once. A per-event framework may not exploit that --- it needs
  measuring, not assuming.
  \medskip
  \item \textbf{Hardware.} A single GPU generation was benchmarked; these are not
  the best numbers the library can produce.
\end{itemize}
\end{frame}

\section{Conclusions}

\begin{frame}{Summary}
\begin{enumerate}
  \item \textbf{Clusters on the GPU and keeps the full merge history}, so
  substructure --- exclusive subjets, soft drop, Lund coordinates --- comes free
  as array reads.
  \medskip
  \item \textbf{Gives the same jets as FastJet.} 150/150 grid points at 100\,\%
  $n_{\rm jets}$ agreement, across \akt/$k_t$/C-A, $R=0.4$--$1.2$ and 24--123
  constituents per jet. $k_t$ and C/A validated on real data for the first time.
  \medskip
  \item \textbf{Reproduces the experiment's own reconstructed jets} exactly, at
  $\sim$600 clusters per event.
  \medskip
  \item \textbf{39--99$\times$ faster} than vectorised FastJet, and the advantage
  \emph{grows} with jet complexity. Cost is independent of $R$ and of algorithm.
  \medskip
  \item Still a \textcolor{fjorange}{\textbf{prototype}}, with meaningful headroom
  left.
\end{enumerate}
\end{frame}

\begin{frame}{What is next}
\begin{center}
\begin{tabular}{@{}clp{0.50\textwidth}@{}}
\toprule
& \textbf{item} & \textbf{why}\\
\midrule
\textbf{1} & C++ FastJet baseline & every number here uses the Python binding;
the production tool is C++\\
\textbf{2} & Tuning & the library has not been optimised for current hardware\\
\textbf{3} & Pile-up, grooming timing & untested axes a realistic workload will
hit\\
\textbf{4} & Use inside experiment software & the real production test\\
\bottomrule
\end{tabular}
\end{center}
\vspace{1em}
\begin{block}{}
\small \textbf{Bottom line:} as an accelerator-resident substitute for CPU
clustering inside a learning workflow, flashjet is \textbf{correct} and roughly
\textbf{two orders of magnitude faster} than per-jet CPU clustering.
\end{block}
\end{frame}
