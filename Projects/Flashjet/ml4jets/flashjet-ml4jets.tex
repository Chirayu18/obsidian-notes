\section{What flashjet is}

\begin{frame}{The problem: clustering is the CPU step in a GPU pipeline}
\begin{columns}[T]
\begin{column}{0.55\textwidth}
Jet clustering sits on the critical path of nearly every jet-level ML pipeline ---
and is almost always done \textbf{on the CPU, off the accelerator}.
\medskip

Four-momenta are copied to the host, clustered, copied back.
\medskip

That round trip dominates when:
\begin{itemize}
  \item training loops revisit the same events for \textbf{many epochs};
  \item jets are \textbf{reclustered} under many systematic or hyper-parameter
        variations;
  \item the clustering itself must sit \textbf{inside} a differentiable pipeline.
\end{itemize}
\end{column}
\begin{column}{0.45\textwidth}
\keybox{\textbf{flashjet} implements the generalised-$k_t$ family
(\akt, $k_t$, Cambridge--Aachen) \textbf{natively on the GPU}.

\smallskip
Data never leaves device memory during clustering.}
\medskip

\smallnote{Written in Triton/PyTorch. One entry point, runs unchanged on CPU and GPU.}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{What it does}
\begin{columns}[T]
\begin{column}{0.52\textwidth}
\textbf{Clusters a batch of jets or events at once}, directly in GPU memory.
\medskip

Returns \textbf{two} things:
\begin{enumerate}
  \item the particle $\rightarrow$ jet assignment;
  \item the \textbf{complete merge history} --- the full binary tree of which pair
        merged, and at what distance.
\end{enumerate}
\end{column}
\begin{column}{0.48\textwidth}
\keybox{\textbf{Why the history matters.}
Substructure observables --- groomed mass, $z_g$, $R_g$, Lund coordinates,
exclusive subjets --- become \textbf{cheap reads of that tree}.

\smallskip
No second clustering pass.}
\end{column}
\end{columns}

\vspace{1em}
\begin{center}
\begin{tabular}{@{}ll@{}}
\toprule
\textbf{Returned} & \textbf{Meaning}\\
\midrule
\texttt{jet\_idx}, \texttt{n\_jets} & particle-to-jet assignment\\
\texttt{hist\_p1}, \texttt{hist\_p2}, \texttt{hist\_child} & which pair merged into what\\
\texttt{hist\_d} & the $d_{ij}$ at each merge step\\
\bottomrule
\end{tabular}
\end{center}
\end{frame}

\begin{frame}[fragile]{The interface}
One function. The algorithm is an argument; the backend is chosen automatically.
\medskip

\begin{Verbatim}[fontsize=\small,frame=single,rulecolor=\color{fjgray}]
out = flashjet.cluster(p4, mask, R=1.0, algorithm="antikt")
#   p4   : (B, N, 4) torch tensor  (px, py, pz, E)
#   mask : (B, N) bool             (ragged -> padded batch)
#   algorithm : "antikt" | "kt" | "cambridge"   (or p=... for generalised-kt)

out.n_jets            # (B,)      jets found per entry
out.jets_p4(p4)       # (B, J, 4) jet four-momenta  (differentiable)
out.splitting_scales()# sqrt(d12), sqrt(d23), ...
out.exclusive_jets(n_jets=3)          # F1
out.groomed_jets(p4, R, z_cut=0.1, beta=0.0)   # F2  soft drop / mMDT
out.lund_coordinates(p4, R)                     # F3  primary Lund plane
\end{Verbatim}

\keybox{Same call on CPU and GPU. \texttt{jets\_p4} is a \texttt{scatter\_add} of
constituents, so gradients flow through the clustering.}
\end{frame}

\begin{frame}{What is implemented}
\begin{columns}[T]
\begin{column}{0.5\textwidth}
\textbf{Clustering (kernels)}
\begin{itemize}
  \item generalised-$k_t$: \akt{} ($p=-1$), $k_t$ ($p=+1$), C/A ($p=0$)
  \item batched, padded/ragged input via a mask
  \item three backends, auto-dispatched:\\
        \smallnote{fused register kernel ($N\!\le\!16$) $\rightarrow$
        scratch NN-array Triton kernel ($N\!\le\!16384$) $\rightarrow$ torch}
  \item full merge history recorded in-kernel
  \item jet-regime (batch of jets) \emph{and} event-regime (whole events)
\end{itemize}
\end{column}
\begin{column}{0.5\textwidth}
\textbf{Substructure --- reads of the history}
\begin{itemize}
  \item \textbf{F1} exclusive $k_t$ subjets \\
        \smallnote{\texttt{exclusive\_jets\_from\_history} --- undo the last merges to
        expose $n$ subjets, or cut at $d_{\rm cut}$}
  \item \textbf{F2} soft drop / mMDT / mass-drop \\
        \smallnote{\texttt{groom\_from\_history} --- walk the harder branch until
        $z > z_{\rm cut}(\Delta R/R)^\beta$}
  \item \textbf{F3} primary Lund coordinates \\
        \smallnote{\texttt{lund\_coordinates\_from\_history} --- $(z,\Delta R,k_t,
        \ln 1/\Delta R, \ln k_t, d)$ per split}
\end{itemize}
\smallskip
\smallnote{All three are \textbf{pure-torch post-reads}: no kernel changes,
CPU/GPU identical, negligible cost.}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Status: a working prototype}
\keybox{\textbf{flashjet is a \prototype.} The physics is validated and the speedups
are real, but it is \textbf{not} production software: it is a Python/Triton library,
not integrated into any experiment's reconstruction framework.}
\medskip

\begin{columns}[T]
\begin{column}{0.5\textwidth}
\textbf{What is solid}
\begin{itemize}
  \item[\textcolor{fjgreen}{\checkmark}] 129 unit tests, incl.\ GPU-only paths
  \item[\textcolor{fjgreen}{\checkmark}] agrees with FastJet jet-by-jet
  \item[\textcolor{fjgreen}{\checkmark}] reproduces analytic predictions
  \item[\textcolor{fjgreen}{\checkmark}] reproduces a published experiment's own
        reconstructed jets
  \item[\textcolor{fjgreen}{\checkmark}] all three algorithms, R-scan, multiplicity scan
\end{itemize}
\end{column}
\begin{column}{0.5\textwidth}
\textbf{What it is not (yet)}
\begin{itemize}
  \item[\textcolor{fjred}{$\times$}] not inside experiment software
        \smallnote{(needs a C++/CUDA rewrite)}
  \item[\textcolor{fjred}{$\times$}] CPU baseline is the \textbf{Python} FastJet
        binding, not C++ FastJet
  \item[\textcolor{fjorange}{$\sim$}] GPU is \textbf{largely idle} --- launch
        configuration tuned on older cards
  \item[\textcolor{fjred}{$\times$}] no pile-up study; grooming not yet timed
\end{itemize}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Caveats --- read the speedups with these}
\begin{enumerate}
  \item \textbf{Hardware.} These are \textbf{V100S} numbers. A current-generation card
  (H100) is roughly a further $2\times$ on identical input, so treat these as a
  \emph{lower bound} --- not as this library's ceiling.
  \medskip
  \item \textbf{The CPU baseline is Python.} Every number compares against FastJet's
  \emph{Python} binding. The production tool is C++. Until that is measured, the honest
  caveat is \emph{``some of this could be Python overhead''}. \textcolor{fjred}{This is
  the single most valuable missing test.}
  \medskip
  \item \textbf{Large-$R$ jets.} These are boosted fat jets ($R\!=\!1.0$-scale,
  $\sim$50--120 constituents), not small-$R$ jets.
  \medskip
  \item \textbf{Batch regime.} The advantage comes from clustering \emph{many} jets at
  once. A per-event framework may not be able to exploit that --- it needs measuring,
  not assuming.
\end{enumerate}
\end{frame}

\section{Downstream: does the history help tagging?}

\begin{frame}{\textcolor{fjorange}{[PLACEHOLDER]} ParT + C/A features --- current status}
\keybox{\textbf{Slide to be finalised.} Numbers below are the current measured
state as of 4 Sep 2026; the follow-up run is still in progress.}
\medskip

\textbf{The idea.} If flashjet hands back the merge tree for free, feed it to the
tagger: add per-particle Cambridge/Aachen branch-point features to Particle
Transformer on JetClass.
\medskip

\textbf{The result: no gain.} Matched 480k checkpoint, 20.05\,M-jet test set:

\begin{center}
\small
\begin{tabular}{@{}lccc@{}}
\toprule
metric & baseline & {} + C/A (5 per-particle) & $\Delta$\\
\midrule
test accuracy & \textbf{85.625\,\%} & 85.524\,\% & $-0.102$\\
mean 1-vs-QCD AUC & \textbf{0.98981} & 0.98970 & $-0.00011$\\
\bottomrule
\end{tabular}
\end{center}
\medskip

\smallnote{All nine classes marginally worse. Magnitudes sit inside the
$\sim$1.3-point seed spread, so the honest statement is
\textbf{``indistinguishable from baseline, with a consistent negative sign''} ---
not ``C/A hurts''. An earlier $+1.3$ at 20k iterations was a \textbf{training
transient}, not a durable effect.}
\end{frame}

\begin{frame}{\textcolor{fjorange}{[PLACEHOLDER]} Why it failed --- and the fix}
\begin{columns}[T]
\begin{column}{0.52\textwidth}
\textbf{Diagnosed, not just observed.}
\smallskip

The feature values are provably correct (0 soft-side violations in 274\,192
entries). The problem is structural:
\medskip

\keybox{\textbf{Measured group size is exactly 1.00} --- the features carry
\emph{zero} prong-grouping information.}
\medskip

{\small In a C/A tree a particle almost always merges into something larger, so
``am I the softer child?'' is true at its \emph{own first merge} --- the walk never
climbs, and the node it lands on is unique to that particle.}
\end{column}
\begin{column}{0.48\textwidth}
\textbf{Consequence.} Prong membership is a property of a \emph{pair}, not of a
particle. The pairwise redesign was built and then \textbf{cancelled unrun}: its
\texttt{share\_bp} channel is identically zero for the same reason.
\medskip

\textbf{Current attempt --- subjet-level features} ($k_t$ cut 20\,GeV):
\smallskip

{\small
\begin{itemize}\setlength{\itemsep}{1pt}
  \item \texttt{part\_sj\_lnm} --- ln mass of my subjet
  \item \texttt{part\_sj\_lnptfrac} --- subjet $p_T$ fraction
  \item \texttt{part\_sj\_nconst} --- constituents sharing it
\end{itemize}}
\smallnote{Passes the set-model probe that killed the earlier variants.
\textbf{Training result pending.}}
\end{column}
\end{columns}
\end{frame}

\section{Conclusions}

\begin{frame}{Summary}
\begin{enumerate}
  \item \textbf{flashjet clusters on the GPU and keeps the full merge history}, so
  substructure --- exclusive subjets, soft drop, Lund coordinates --- comes free as
  array reads rather than a second clustering pass.
  \medskip
  \item \textbf{It gives the same jets as FastJet.} 150/150 grid points at
  100\,\% $n_{\rm jets}$ agreement, across \akt{}/$k_t$/C-A, $R=0.4$--$1.2$, and
  24--123 constituents per jet. $k_t$ and C/A validated on real data for the first time.
  \medskip
  \item \textbf{It reproduces ATLAS's own reconstructed jets} exactly, clustering
  $\sim$600 calorimeter clusters per event.
  \medskip
  \item \textbf{39--99$\times$ faster} than vectorised FastJet on a V100S, and the
  advantage \emph{grows} with jet complexity. Cost is independent of $R$ and of the
  algorithm choice.
  \medskip
  \item \textbf{Still a \prototype{}}, with meaningful headroom: the profiler says the
  GPU is largely idle and attributes $\sim$1.7--1.8$\times$ to register pressure alone.
\end{enumerate}
\end{frame}

the GPU is idle; retuning launch geometry is free speed\\
\textbf{3} & \textcolor{fjorange}{Pile-up and grooming timing} & untested axes that a
realistic workload will hit\\
\textbf{4} & \textcolor{fjred}{Use it inside experiment software} & the real production
test --- needs a C++/CUDA rewrite\\
\bottomrule
\end{tabular}
\end{center}
\vspace{1em}
\keybox{\textbf{Bottom line:} as an accelerator-resident substitute for CPU clustering
inside a learning workflow, flashjet is \textbf{correct}, roughly \textbf{two orders of
magnitude faster} than per-jet FastJet, and still leaving most of the GPU idle.}
\end{frame}

\begin{frame}{Backup --- why not PHYSLITE?}
ATLAS's main research open-data format, \textbf{DAOD\_PHYSLITE}, cannot be used for this
benchmark: it stores calibrated \emph{objects}, not jet-finding \emph{inputs}.
\medskip

\begin{center}

\toprule
looked for & found\\
\midrule

Flow-element / PFO containers & 0 branches\\
general \texttt{TruthParticles} & absent\\
\texttt{constituentLinks} on jets & present but \textbf{dangling}\\
\texttt{InDetTrackParticles} & thinned to $\sim$4 tracks/event\\
\bottomrule
\end{tabular}
\end{center}
\medskip

\smallnote{The links carry a \emph{non-zero} container key with 22.5 constituents/jet
recorded --- the structure is described, but the four-vectors are not shipped. ATLAS
confirm this is deliberate: \emph{``we don't store jet constituents in PHYSLITE.''}
Hence the datasets used here, which do ship constituents.}
