\section{Jet clustering, and flashjet}

\begin{frame}{Reconstruction is moving to accelerators --- clustering has not}
\begin{columns}[T]
\begin{column}{0.54\textwidth}
Reconstruction is steadily becoming \textbf{heterogeneous}: as much of the event
as possible is expected to run on the accelerator.
\medskip

CMS has run GPUs in the High-Level Trigger since the start of Run~3 ---
\textbf{$\sim$30\,\% of HLT reconstruction} is offloaded (pixel local
reconstruction, pixel-only tracks and vertices, calorimeter local
reconstruction), for a \textbf{$\sim$25\,\% reduction} in total HLT time.
\medskip

\smallnote{Tracking, vertexing and calorimetry have all been ported.
\textbf{Jet clustering has not} --- it is one of the few steps still bound to the
CPU. \tiny{[CMS HLT, Run~3]}}
\end{column}
\begin{column}{0.46\textwidth}
\begin{block}{Why it is still on the CPU}
Sequential recombination is \textbf{IRC safe} --- which is why it survived --- but
each merge \textbf{depends on the one before it}. That data dependence is exactly
what does not map onto a GPU in the obvious way.
\end{block}
\medskip
\begin{block}{Where it hurts today}
\small
\begin{itemize}
  \item reclustering jets \textbf{inside a training loop}
  \item scanning the \textbf{jet radius}
  \item accessing \textbf{substructure} dynamically
\end{itemize}
Each pays a \textbf{host\,$\leftrightarrow$\,device transfer}, over and over.
\end{block}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{flashjet}
\begin{columns}[T]
\begin{column}{0.53\textwidth}
An \textbf{open-source} library bringing $k_t$, Cambridge--Aachen and \akt{}
natively to the GPU through custom \textbf{Triton} kernels.
\medskip

\textbf{Why Triton.} It is the GPU language of the PyTorch ecosystem, so the
kernel \textbf{compiles and autotunes for the hardware at hand} and drops into
tensor pipelines directly --- with \textbf{no separate CUDA build} and no
hand-written device code to maintain.
\medskip

\textbf{Execution model.} One kernel program clusters \textbf{one event entirely
on chip}; the whole batch goes out in a \textbf{single fused launch}. Nothing
returns to the host.
\medskip

\smallnote{Jets, clustering histories and derived substructure all come back as
\textbf{tensors}.}
\end{column}
\begin{column}{0.47\textwidth}
\begin{block}{What makes it tractable}
\small The Cacciari--Salam nearest-neighbour lemma: the global $d_{ij}$ minimum
is always realised at some particle's \textbf{geometric} nearest neighbour. Each
slot carries its own NN, so a merge invalidates only $O(1)$ rows on average ---
\textbf{$O(N^2)$ per event instead of $O(N^3)$}.
\end{block}
\smallskip
\begin{block}{Engineering}
\small
\begin{itemize}\setlength{\itemsep}{1pt}
  \item bandwidth-bound: the NN update and the \textbf{next argmin} are fused
        into one sweep
  \item \textbf{branchless} pair-vs-beam, via masked stores
  \item launch config autotuned \textbf{once per GPU model} and cached, so runs
        stay \textbf{reproducible}
\end{itemize}
\end{block}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{How sequential recombination works}
\begin{center}
\includegraphics[width=0.93\linewidth]{fig/clustering_steps.png}
\end{center}
\vspace{-0.7em}
\smallnote{Repeat until nothing is left: if the smallest distance is a $d_{ij}$,
merge that pair; if it is a $d_{iB}$, call that a jet and remove it. One exponent
$p$ selects the algorithm --- the \emph{same} kernel serves all three.}
\end{frame}

\begin{frame}{One exponent, three algorithms}
\begin{columns}[T]
\begin{column}{0.55\textwidth}
The generalised-$k_t$ distance measure:
\begin{align*}
d_{ij} &= \min\!\left(p_{Ti}^{2p},\, p_{Tj}^{2p}\right)
          \frac{\Delta R_{ij}^{2}}{R^{2}},
&\Delta R_{ij}^{2} &= (y_i-y_j)^2+(\phi_i-\phi_j)^2\\[2pt]
d_{iB} &= p_{Ti}^{2p}
\end{align*}
\vspace{-0.6em}

Merge the pair with the smallest $d_{ij}$; if the smallest distance is a
$d_{iB}$, call $i$ a jet. \textbf{Only $p$ changes:}

\smallskip
\small
\begin{tabular}{@{}cll@{}}
\toprule
$p$ & algorithm & merges first\\
\midrule
$-1$ & \akt & around the \textbf{hardest} particles\\
$0$ & Cambridge--Aachen & the \textbf{closest} pair\\
$+1$ & $k_t$ & the \textbf{softest} pair\\
\bottomrule
\end{tabular}
\end{column}
\begin{column}{0.45\textwidth}
\includegraphics[width=0.97\linewidth]{fig/jet_areas.png}\\
\smallnote{\akt{} catchment areas $\to \pi R^2$ --- the defining rigid-cone
result, here from flashjet's own clustering.}
\medskip

\textbf{\small The ordering is what the substructure reads:}
\smallskip
{\small
\begin{itemize}\setlength{\itemsep}{1pt}
  \item \textbf{C/A} by \emph{angle} $\rightarrow$ \textbf{grooming}, \textbf{Lund}
  \item \textbf{$k_t$} by \emph{relative $p_T$} $\rightarrow$ \textbf{exclusive
        subjets}
\end{itemize}}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{It returns the tree, not just the jets}
Clustering a batch returns \textbf{two} things --- the jets, and the record of how
they were built.
\medskip

\begin{center}
\small
\begin{tabular}{@{}lll@{}}
\toprule
\textbf{field} & \textbf{shape} & \textbf{what it holds}\\
\midrule
\texttt{jet\_idx} & $(B,N)$ & which jet each particle ended up in\\
\texttt{n\_jets} & $(B,)$ & how many jets were found\\
\addlinespace
\texttt{hist\_p1}, \texttt{hist\_p2} & $(B,N)$ & \textbf{which pair merged}, at each step\\
\texttt{hist\_child} & $(B,N)$ & what they merged \emph{into}\\
\texttt{hist\_d} & $(B,N)$ & the distance $d_{ij}$ at that step\\
\bottomrule
\end{tabular}
\end{center}
\medskip

\begin{block}{Why the second half matters}
\small The bottom three rows are the \textbf{complete binary merge tree}. Groomed
mass, $z_g$, $R_g$, Lund coordinates and exclusive subjets are all
\textbf{coordinates on that tree} --- so they become array reads, with
\textbf{no second clustering pass}.
\end{block}
\end{frame}

\begin{frame}[fragile]{Using it}
\begin{Verbatim}[fontsize=\footnotesize,frame=single,rulecolor=\color{fjgray}]
import flashjet

out = flashjet.cluster(p4, mask, R=1.0, algorithm="antikt")
\end{Verbatim}
\vspace{-0.3em}
{\footnotesize
\begin{tabular}{@{}lll@{}}
\toprule
\textbf{argument} & \textbf{type} & \\
\midrule
\texttt{p4} & $(B,N,4)$ tensor & $(p_x,p_y,p_z,E)$; CPU or CUDA\\
\texttt{mask} & $(B,N)$ bool & marks real constituents in a padded batch\\
\texttt{R} & float & jet radius\\
\texttt{algorithm} & str & \texttt{"antikt"} $\vert$ \texttt{"kt"} $\vert$ \texttt{"cambridge"}\\
\texttt{p} & float & generalised-$k_t$ exponent (overrides \texttt{algorithm})\\
\bottomrule
\end{tabular}}
\medskip

\begin{Verbatim}[fontsize=\footnotesize,frame=single,rulecolor=\color{fjgray}]
out.n_jets                                    # (B,)     jets found
out.jets_p4(p4)                               # (B,J,4)  jet four-momenta
out.splitting_scales()                        #          sqrt(d12), sqrt(d23), ...
out.exclusive_jets(n_jets=3)                  # F1       exclusive kt subjets
out.groomed_jets(p4, R, z_cut=0.1, beta=0.0)  # F2       soft drop / mMDT
out.lund_coordinates(p4, R)                   # F3       primary Lund plane
\end{Verbatim}
\smallnote{The same call runs unchanged on CPU and GPU --- the backend is chosen
from the problem size, not by the caller.}
\end{frame}

\begin{frame}{What is implemented}
\begin{columns}[T]
\begin{column}{0.5\textwidth}
\textbf{Clustering}
\begin{itemize}
  \item generalised-$k_t$: \akt, $k_t$, C/A
  \item batched, ragged input via a mask
  \item three backends, auto-dispatched by size
  \item merge history recorded in-kernel
  \item jet regime \emph{and} event regime
\end{itemize}
\end{column}
\begin{column}{0.5\textwidth}
\textbf{Substructure} --- reads of the history
\begin{itemize}
  \item \textbf{F1} exclusive $k_t$ subjets
  \item \textbf{F2} soft drop / mMDT / mass-drop
  \item \textbf{F3} primary Lund coordinates
\end{itemize}
\smallskip
\smallnote{All three are pure-torch post-reads: no kernel changes, CPU/GPU
identical, negligible cost.}
\end{column}
\end{columns}
\end{frame}

% ------------------------------------------------------------
\begin{frame}{F1 --- exclusive subjets from the $k_t$ history}
\begin{columns}[T]
\begin{column}{0.42\textwidth}
\textbf{What it does.} Undo the last merges of the sequence to expose exactly
$n$ subjets --- or stop at a $d_{cut}$.
\smallskip

\textbf{How.} The history is already a binary tree. Walk back up it $n-1$ steps;
no reclustering.
\smallskip

\textbf{Why $k_t$.} $k_t$ merges the softest pair first, so the \emph{last}
merges are the hard prongs --- exactly what a 2-prong decay leaves behind.
\smallskip

The splitting scales come straight from \texttt{hist\_d}:
\[\sqrt{d_{12}},\ \sqrt{d_{23}},\ \dots\ \ \text{with}\ \
d_{ij}=\min(p_{Ti}^{2},p_{Tj}^{2})\,\Delta R_{ij}^2/R^2\]
\end{column}
\begin{column}{0.58\textwidth}
\includegraphics[width=\linewidth]{fig/kt_observables.png}\\
\smallnote{Toy W-like (2-prong) vs QCD-like fat jets. \textbf{Left:} the $k_t$
splitting scale $\sqrt{d_{12}}$ separates cleanly. \textbf{Right:} the momentum
balance $z$ of the two exclusive subjets --- the W sits near $z\!\sim\!0.4$,
QCD spreads to small $z$.}
\end{column}
\end{columns}
\end{frame}

\begin{columns}[T]

\smallnote{Walk down the C/A tree along the harder branch, discarding the softer

\begin{frame}{F3 --- primary Lund coordinates from the C/A history}
\begin{center}
\includegraphics[width=0.80\linewidth]{fig/lund_plane.png}
\end{center}
\vspace{-0.6em}
\begin{columns}[T]
\begin{column}{0.34\textwidth}
\smallnote{\textbf{What.} Every primary splitting emits
$z=\tfrac{p_{T2}}{p_{T1}+p_{T2}}$, $\Delta R$, $k_t=p_{T2}\Delta R$, and the

scales computed independently.}
\end{column}

\end{column}
\begin{column}{0.5\textwidth}
\textbf{Method}
\begin{enumerate}
  \item extract constituents once, save;
  \item cluster on GPU, time it, save the \emph{same} events;
  \item cluster those saved events with FastJet;
  \item \textbf{check they agree} --- else the timing is meaningless.
\end{enumerate}
\smallskip
\begin{block}{}
\small Both clusterers read the \textbf{same saved file}. Speedups are quoted
against the \textbf{vectorised} FastJet interface --- the faster, fairer baseline.
\end{block}
\end{column}

\section{Conclusions}

\begin{frame}{Summary}
\begin{enumerate}
  \item \textbf{Clusters on the GPU and keeps the full merge history}, so
  substructure --- exclusive subjets, soft drop, Lund coordinates --- comes free
  as array reads.
  \medskip
  \item \textbf{Gives the same jets as FastJet.} 150/150 grid points at 100\,\%
  $n_{\rm jets}$ agreement, across \akt/$k_t$/C-A, $R=0.4$--$1.2$ and 24--123
  constituents per jet. $k_t$ and C/A validated on real data for the first time.
  \medskip
  \item \textbf{Reproduces the experiment's own reconstructed jets} exactly, at
  $\sim$600 clusters per event.
  \medskip
  \item \textbf{39--99$\times$ faster} than vectorised FastJet, and the advantage
  \emph{grows} with jet complexity. Cost is independent of $R$ and of algorithm.
  \medskip
  \item Still a \textcolor{fjorange}{\textbf{prototype}}, with meaningful headroom
  left.
\end{enumerate}
\end{frame}
