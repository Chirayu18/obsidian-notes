\begin{frame}{RAGAS factual correctness disagrees with the rubric}
  \vspace{2pt}
  \begin{columns}[T]
    \column{0.47\textwidth}
    \small \texttt{gmn-control-region-1}, mean of 3 runs per configuration
    \vspace{4pt}
    \begin{center}
    \begin{tabular}{lrr}
      \toprule
      agent & FC recall & rubric (1--5) \\
      \midrule
      GPT-5.6 luna, 3 configs  & 0.43--0.55 & 5.0 \\
      GPT-5.6 terra, 3 configs & 0.41--0.66 & 5.0 \\
      Qwen 3.8, read-only config & 0.55 & 3.7 \\
      \bottomrule
    \end{tabular}
    \end{center}
    \small
    \begin{itemize}
      \item 37 answers got rubric 5/5; their FC ranges \con{0.22 to 0.92}
      \item reading them: the GPT answers are correct
    \end{itemize}
    \column{0.49\textwidth}
    \small What \texttt{FactualCorrectness(mode="recall")} computes\\
    \grey{(ragas 0.2.15, \texttt{\_factual\_correctness.py})}
    \[
      \text{recall}=\frac{TP}{TP+FN}
    \]
    \begin{itemize}
      \item $TP$: claims split from the \textbf{answer} that the gold supports
      \item $FN$: claims split from the \textbf{gold} missing in the answer
      \item two different LLM splits, \con{redone on every call}
      \item every gold claim weighs the same, side remarks included\\
        \grey{(e.g.\ the docs' ``$\sim$50\%'' comment in gmn)}
    \end{itemize}
  \end{columns}
  \takeaway{Not ``fraction of gold covered'': it moves with answer length and claim split}
\end{frame}

% -------------------------------------------------------------------
\begin{frame}{Replacement: per-question key points (``nuggets'')}
  \vspace{2pt}
  \begin{columns}[T]
    \column{0.52\textwidth}
    \small Written once per question, frozen, versioned (\texttt{nuggets.yaml}):
    \vspace{3pt}
    \begin{center}
    \begin{tabular}{@{}l>{\raggedright\arraybackslash}p{0.72\linewidth}@{}}
      \toprule
      \multicolumn{2}{@{}l}{\texttt{gmn-control-region-1}} \\
      \midrule
      \pos{vital} & use the \texttt{gmN} uncertainty type \\
      \pos{vital} & CR count $N$ written after the keyword \\
      \pos{vital} & extrapolation factor $\alpha$ in the process column \\
      \pos{vital} & rate $= N\alpha$ \\
      \grey{okay} & concrete datacard line; $1/\sqrt{N+1}\approx45\%$ \\
      \con{must\_not} & recommends \texttt{lnN}/\texttt{lnU}; example card inconsistent \\
      \bottomrule
    \end{tabular}
    \end{center}
    \column{0.44\textwidth}
    \small Judge (Qwen 3.8 on aigw) per item:
    \begin{itemize}
      \item supported / partly / not supported
      \item must quote the answer; harness checks the quote is really there
      \item gold prose not shown; items shuffled; 3 judge runs, majority
    \end{itemize}
    Scores per answer:
    \[
      V_\text{strict}=\frac{\#\,\text{vital fully supported}}{\#\,\text{vital}}
    \]
    \textbf{pass} $=$ all vital supported \textbf{and} no \con{must\_not} present
  \end{columns}
  \takeaway{Method of the TREC 2024/25 RAG track (AutoNuggetizer), plus evidence quotes and must\_not items}
\end{frame}

\begin{frame}{What the nuggets add: a named reason for every failure}
  \vspace{2pt}
  \begin{columns}[T]
    \column{0.49\textwidth}
    \textbf{Caught, rubric said 5/5}
    \small
    \begin{itemize}
      \item terra shipped config, gmn run 3: example card\\
        \texttt{bkg\_stat gmN - 0.12} \con{(no $N$ after gmN)}\\
        $\to$ \con{must\_not} ``inconsistent example'' present
    \end{itemize}
    \textbf{Diagnosed}
    \begin{itemize}
      \item killed luna runs (permission reject, path typo): miss both vital points, not ``low recall''
      \item Qwen read-only, shape run 2: ``shape and counting cards cannot be combined'' $\to$ \con{must\_not}
    \end{itemize}
    \column{0.47\textwidth}
    \textbf{Missed / wrong (judge limits)}
    \small
    \begin{itemize}
      \item terra no-web config, gmn run 3: \texttt{rate 10.0} with \texttt{gmN 25 0.12} ($25\times0.12=3.0$): arithmetic not caught
      \item 2 correct shape-counting answers failed: judge padded its quote with text not in the answer, quote check rejected it
